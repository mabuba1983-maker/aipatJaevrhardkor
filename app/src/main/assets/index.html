let lastLibraryError='';
function showLibraryError(error, source='runtime'){
  const box=$('errorDock'), code=$('errorCode');
  if(!box||!code) return;
  const err=error instanceof Error ? error : null;
  const message=err ? (err.name+': '+err.message) : String(error??'Unknown error');
  const stack=err&&err.stack ? err.stack : '';
  const id='LYL-ERR-'+Date.now().toString(36).toUpperCase();
  lastLibraryError='Error ID: '+id+'\\nSource: '+source+'\\n'+message+(stack?'\\nStack:\\n'+stack:'');
  code.textContent=lastLibraryError;
  box.classList.add('show');
}
$('errorClose').addEventListener('click',()=> $('errorDock').classList.remove('show'));
$('errorCopy').addEventListener('click',async()=>{
  try{
    if(navigator.clipboard&&navigator.clipboard.writeText) await navigator.clipboard.writeText(lastLibraryError);
    else{
      const t=document.createElement('textarea'); t.value=lastLibraryError;
      t.style.position='fixed'; t.style.opacity='0'; document.body.appendChild(t); t.select();
      document.execCommand('copy'); t.remove();
    }
    $('errorCopy').textContent='تم النسخ';
    setTimeout(()=>{ if($('errorCopy')) $('errorCopy').textContent='نسخ تفاصيل الخطأ'; },1400);
  }catch(e){ $('errorCopy').textContent='تعذر النسخ — حدّد النص وانسخه'; }
});
window.addEventListener('error',e=>{
  const detail=e.error||((e.message||'Unknown error')+'\\n'+(e.filename||'')+':'+(e.lineno||0)+':'+(e.colno||0));
  showLibraryError(detail,'window.error');
});
window.addEventListener('unhandledrejection',e=>showLibraryError(e.reason,'unhandledrejection'));

function setConnected(on){
  connected=on;
  const bar=$('connBar');
  if(on){ bar.className='on'; bar.textContent='🟢 متصل بالتلفزيون — جاهز للإضافة'; }
  else { bar.className='off'; bar.textContent='🔴 غير متصل — اضغط هنا للاتصال بالتلفزيون'; }
}
function openConnModal(){ $('connModal').style.display='flex'; $('connMsg').textContent=''; }
function closeConnModal(){ $('connModal').style.display='none'; }

function connectToTv(code){
  code = code || $('connCodeInput').value.trim();
  if(!/^[0-9]{6}$/.test(code)){ $('connMsg').textContent='⚠️ الرمز لازم يكون 6 أرقام'; return; }
  $('connMsg').textContent='جاري الاتصال...';
  try{ if(peer) peer.destroy(); }catch(e){ showLibraryError(e,'closing previous PeerJS connection'); }
  peer=new Peer(undefined,{config:{iceServers:[{urls:'stun:stun.l.google.com:19302'},{urls:'stun:stun1.l.google.com:19302'}]}});
  peer.on('open',()=>{
    conn=peer.connect('wrglib'+code,{reliable:true});
    conn.on('open',()=>{ setConnected(true); closeConnModal(); });
    conn.on('data',d=>{ try{ const m=(typeof d==='string')?JSON.parse(d):d; if(m.t==='ack'){ /* يمكن استخدامها لاحقًا لتتبع التقدم بدقة */ } }catch(e){ showLibraryError(e,'PeerJS data parsing'); } });
    conn.on('close',()=>{ setConnected(false); });
    conn.on('error',e=>{ showLibraryError(e,'PeerJS connection'); $('connMsg').textContent='⚠️ تعذر الاتصال، راجع تفاصيل الخطأ'; setConnected(false); });
  });
  peer.on('error',e=>{ showLibraryError(e,'PeerJS peer'); $('connMsg').textContent='⚠️ تعذر الاتصال، راجع تفاصيل الخطأ'; });
}

const CHUNK_SIZE=16000;
function sendGameToTv(game,btn,pctEl){
  if(!connected||!conn||!conn.open){ openConnModal(); return; }
  if(!game.b64){ alert('هذي اللعبة ما فيها محتوى (b64) بعد'); return; }
  btn.disabled=true; btn.textContent='جاري الإرسال...';
  const total=Math.ceil(game.b64.length/CHUNK_SIZE);
  conn.send(JSON.stringify({t:'meta',name:game.name,icon:game.icon,totalChunks:total}));
  let i=0;
  function sendNext(){
    if(i>=total){ btn.textContent='✅ أُضيفت'; pctEl.textContent=''; return; }
    const piece=game.b64.slice(i*CHUNK_SIZE,(i+1)*CHUNK_SIZE);
    try{ conn.send(JSON.stringify({t:'chunk',i,data:piece})); }catch(e){ showLibraryError(e,'sending game chunk '+i); btn.disabled=false; btn.textContent='إعادة المحاولة'; return; }
    i++;
    pctEl.textContent=Math.round(i/total*100)+'%';
    setTimeout(sendNext,12); /* فاصل بسيط بين كل جزء وجزء عشان ما نطفّح قناة البيانات */
  }
  sendNext();
}

let qrScanStream=null, qrScanRAF=null;
async function openQrScan(){
  if(typeof jsQR==='undefined'){ $('connMsg').textContent='⚠️ تعذر تحميل ماسح الباركود'; return; }
  $('qrScanOverlay').style.display='flex';
  try{
    qrScanStream=await navigator.mediaDevices.getUserMedia({video:{facingMode:'environment'}});
    const v=$('qrScanVideo'); v.srcObject=qrScanStream; await v.play();
    qrScanTick();
  }catch(e){ showLibraryError(e,'QR camera'); $('connMsg').textContent='⚠️ ما قدرنا نفتح الكاميرا'; closeQrScan(); }
}
function qrScanTick(){
  const v=$('qrScanVideo'), c=$('qrScanCanvas');
  if(v&&v.readyState===v.HAVE_ENOUGH_DATA){
    c.width=v.videoWidth; c.height=v.videoHeight;
    const ctx=c.getContext('2d'); ctx.drawImage(v,0,0,c.width,c.height);
    try{
      const img=ctx.getImageData(0,0,c.width,c.height);
      const res=jsQR(img.data,img.width,img.height);
      if(res&&res.data){
        const digits=(res.data.match(/\d{6}/)||[null])[0];
        if(digits){ closeQrScan(); connectToTv(digits); return; }
      }
    }catch(e){ showLibraryError(e,'QR frame decoding'); }
  }
  qrScanRAF=requestAnimationFrame(qrScanTick);
}
function closeQrScan(){
  if(qrScanRAF) cancelAnimationFrame(qrScanRAF); qrScanRAF=null;
  if(qrScanStream){ qrScanStream.getTracks().forEach(t=>t.stop()); qrScanStream=null; }
  $('qrScanOverlay').style.display='none';
}
