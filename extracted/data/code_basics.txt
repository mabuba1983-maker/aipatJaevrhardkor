# Python basics
def square(n):
    return n * n

assert square(4) == 16

# Loop and sum
total = 0
for i in range(1, 6):
    total += i
assert total == 15

# Dictionary count
words = "a b a".split()
count = {}
for w in words:
    count[w] = count.get(w, 0) + 1
assert count == {"a": 2, "b": 1}
