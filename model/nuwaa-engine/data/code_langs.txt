// Kotlin
fun sum(a: List<Int>): Int = a.sum()

// Java
public class Main {
    static int sum(int[] a) { int s = 0; for (int x : a) s += x; return s; }
    public static void main(String[] args) { System.out.println(sum(new int[]{1, 2, 3})); }
}
